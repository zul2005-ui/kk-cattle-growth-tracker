    <script>
        const ctx = document.getElementById('growthChart').getContext('2d');
        const resultPanel = document.getElementById('resultPanel');
        
        // Mathematical Constants derived from Malaysian MARDI growth models for KK Cattle
        const W_base = 50;   // Weaning weight (kg)
        const W_max = 250;  // Maximum capacity gain (kg)
        const k = 0.015;    // Growth rate constant
        const x_mid = 180;  // Inflection point in days (around 6 months post-weaning)

        // Generate data points for the 0 to 400 days range
        const labels = [];
        const weightData = [];
        const derivativeData = [];

        for (let x = 0; x <= 400; x += 10) {
            labels.push(x);
            
            // Logistic Growth Function f(x)
            const fx = W_base + (W_max / (1 + Math.exp(-k * (x - x_mid))));
            weightData.push(Number(fx.toFixed(2)));

            // Derivative Function f'(x)
            const expTerm = Math.exp(-k * (x - x_mid));
            const fPrimeX = (W_max * k * expTerm) / Math.pow(1 + expTerm, 2);
            derivativeData.push(Number(fPrimeX.toFixed(3)));
        }

        // Initialize Chart.js with custom animations
        const growthChart = new Chart(ctx, {
            type: 'line',
            data: {
                labels: labels,
                datasets: [
                    {
                        label: 'Cattle Weight f(x) (kg)',
                        data: weightData,
                        borderColor: '#10b981',
                        backgroundColor: 'rgba(16, 185, 129, 0.1)',
                        borderWidth: 3,
                        yAxisID: 'yWeight',
                        tension: 0.2,
                        pointRadius: 2,
                        pointHoverRadius: 8,
                        pointHoverBackgroundColor: '#10b981'
                    },
                    {
                        label: "Growth Rate f'(x) (kg/day)",
                        data: derivativeData,
                        borderColor: '#f59e0b',
                        borderDash: [5, 5],
                        borderWidth: 2,
                        yAxisID: 'yRate',
                        tension: 0.2,
                        pointRadius: 0 // Hidden until hovered/clicked
                    }
                ]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                animation: {
                    duration: 2000,
                    easing: 'easeOutQuart'
                },
                scales: {
                    x: {
                        title: { display: true, text: 'Days Post-Weaning', color: '#94a3b8' },
                        grid: { color: 'rgba(148, 163, 184, 0.1)' },
                        ticks: { color: '#94a3b8' }
                    },
                    yWeight: {
                        type: 'linear',
                        position: 'left',
                        title: { display: true, text: 'Total Weight (kg)', color: '#10b981' },
                        grid: { color: 'rgba(148, 163, 184, 0.1)' },
                        ticks: { color: '#10b981' },
                        min: 40,
                        max: 320
                    },
                    yRate: {
                        type: 'linear',
                        position: 'right',
                        title: { display: true, text: 'Daily Rate of Change (kg/day)', color: '#f59e0b' },
                        grid: { drawOnChartArea: false },
                        ticks: { color: '#f59e0b' },
                        min: 0,
                        max: 1.2
                    }
                },
                plugins: {
                    legend: {
                        labels: { color: '#f1f5f9' }
                    }
                },
                // Click handler configuration
                onClick: (e, elements) => {
                    if (elements.length > 0) {
                        const index = elements[0].index;
                        const day = labels[index];
                        const weight = weightData[index];
                        const rate = derivativeData[index];
                        
                        displayCalculusInterpretation(day, weight, rate);
                    }
                }
            }
        });

        // Function to compute livestock interpretation based on derivative values
        function displayCalculusInterpretation(day, weight, rate) {
            document.getElementById('resDay').innerText = `Day ${day}`;
            document.getElementById('resWeight').innerText = `${weight} kg`;
            document.getElementById('resRate').innerText = `${rate} kg/day`;

            let laymanExplanation = "";

            if (day < 120) {
                laymanExplanation = `The calf is starting steady structural frame development. The derivative is small but expanding ($f'(x)$ is accelerating), indicating the feed is efficiently building early muscle mass.`;
            } else if (day >= 120 && day <= 240) {
                laymanExplanation = `Peak performance zone! The derivative is at its absolute maximum near Day 180 ($f'(x) \\approx ${rate}$ kg/day). The animal is converting feed to beef at maximum genetic potential. Maintain intensive feed ratios.`;
            } else {
                laymanExplanation = `Growth velocity is slowing down ($f'(x)$ approaches zero). The cattle has reached operational biological maturity. Keeping the animal past this curve reduces profitability as feed cost will outweigh marginal weight gain.`;
            }

            // Reveal and animate the result module
            resultPanel.classList.remove('hidden');
            setTimeout(() => {
                resultPanel.classList.remove('scale-95');
                resultPanel.classList.add('scale-100');
            }, 50);
        }
    </script>
